cmake_minimum_required(VERSION 3.16)
project(RosPocket LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(CycloneDDS REQUIRED)
add_library(pocket_core STATIC inspector.cpp generated/PocketTypes.c)
set_target_properties(pocket_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(pocket_core PUBLIC . generated)
target_link_libraries(pocket_core PUBLIC CycloneDDS::ddsc)
target_compile_options(pocket_core PRIVATE -Wall -Wextra)
if(ANDROID)
 add_library(rospocket SHARED jni.cpp)
 target_link_libraries(rospocket PRIVATE pocket_core log)
 target_link_options(rospocket PRIVATE -Wl,-z,max-page-size=16384)
else()
 enable_testing()
 set(POCKET_TEST_ADDRESS "127.0.0.1" CACHE STRING "LAN IPv4 for the existing isolated domain-67 fixture")
 add_executable(subscribe_failure_test ../../../../tests/subscribe_failure_test.cpp)
 target_link_libraries(subscribe_failure_test pocket_core)
 add_test(NAME subscribe_failure COMMAND subscribe_failure_test "${POCKET_TEST_ADDRESS}")
 set_tests_properties(subscribe_failure PROPERTIES TIMEOUT 45 LABELS "fixture;domain67")
 add_executable(session_test ../../../../tests/session_test.cpp)
 target_link_libraries(session_test pocket_core)
 add_executable(probe ../../../../tests/probe.cpp)
 target_link_libraries(probe pocket_core)
endif()
if(NOT ANDROID)
 add_test(NAME subscribe_failure_loopback COMMAND subscribe_failure_test --loopback)
 set_tests_properties(subscribe_failure_loopback PROPERTIES TIMEOUT 45 LABELS "fixture;isolated;domain67" RUN_SERIAL TRUE)
 add_executable(connection_state_test ../../../../tests/connection_state_test.cpp)
 target_include_directories(connection_state_test PRIVATE .)
 add_test(NAME connection_state COMMAND connection_state_test)
 add_test(NAME session COMMAND session_test "127.0.0.1")
 add_executable(core_test ../../../../tests/core_test.cpp)
 target_include_directories(core_test PRIVATE .)
 add_test(NAME core COMMAND core_test)
 add_executable(connection_fixture_test ../../../../tests/connection_fixture_test.cpp)
 target_link_libraries(connection_fixture_test pocket_core)
 add_test(NAME connection_fixture COMMAND connection_fixture_test)
 set_tests_properties(connection_fixture PROPERTIES TIMEOUT 90 LABELS "fixture;isolated;domain69" RUN_SERIAL TRUE)
endif()
